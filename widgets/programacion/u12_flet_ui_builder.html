Interactúa con la ventana reactiva y observa la estructura declarativa de controles y eventos.

Ventana Flet Reactiva
¡Hola!
import flet as ft
def main(page: ft.Page):
  page.title = "App Flet"
  txt_nombre = ft.TextField(label="Nombre")
  lbl_saludo = ft.Text()
  def saludar(e):
    lbl_saludo.value = f"¡Hola {txt_nombre.value}!"
    page.update()
  page.add(txt_nombre, ft.ElevatedButton("Saludar", on_click=saludar), lbl_saludo)
ft.app(target=main)
